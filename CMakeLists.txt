cmake_minimum_required(VERSION 3.20)

project(TribesAscendServerLagCompensation CXX)

if(NOT "RelWithDebInfoAndErrorChecks" IN_LIST CMAKE_CONFIGURATION_TYPES)
    list(APPEND CMAKE_CONFIGURATION_TYPES RelWithDebInfoAndErrorChecks)
    list(APPEND CMAKE_CONFIGURATION_TYPES RelWithDebInfoAndErrorChecksAndAsserts)
    list(REMOVE_DUPLICATES CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_CONFIGURATION_TYPES "${CMAKE_CONFIGURATION_TYPES}" CACHE STRING "" FORCE)
endif()

# Custom build type: RelWithDebInfo plus PERFORM_ERROR_CHECKS
set(CMAKE_CXX_FLAGS_RELWITHDEBINFOANDERRORCHECKS "${CMAKE_CXX_FLAGS_RELWITHDEBINFO}")
set(CMAKE_SHARED_LINKER_FLAGS_RELWITHDEBINFOANDERRORCHECKS "${CMAKE_SHARED_LINKER_FLAGS_RELWITHDEBINFO}")
set(CMAKE_EXE_LINKER_FLAGS_RELWITHDEBINFOANDERRORCHECKS "${CMAKE_EXE_LINKER_FLAGS_RELWITHDEBINFO}")
set(CMAKE_MODULE_LINKER_FLAGS_RELWITHDEBINFOANDERRORCHECKS "${CMAKE_MODULE_LINKER_FLAGS_RELWITHDEBINFO}")
string(APPEND CMAKE_CXX_FLAGS_RELWITHDEBINFOANDERRORCHECKS " /DPERFORM_ERROR_CHECKS")

# Custom build type: RelWithDebInfo minus NDEBUG plus PERFORM_ERROR_CHECKS
set(CMAKE_CXX_FLAGS_RELWITHDEBINFOANDERRORCHECKSANDASSERTS "${CMAKE_CXX_FLAGS_RELWITHDEBINFO}")
set(CMAKE_SHARED_LINKER_FLAGS_RELWITHDEBINFOANDERRORCHECKSANDASSERTS "${CMAKE_SHARED_LINKER_FLAGS_RELWITHDEBINFO}")
set(CMAKE_EXE_LINKER_FLAGS_RELWITHDEBINFOANDERRORCHECKSANDASSERTS "${CMAKE_EXE_LINKER_FLAGS_RELWITHDEBINFO}")
set(CMAKE_MODULE_LINKER_FLAGS_RELWITHDEBINFOANDERRORCHECKSANDASSERTS "${CMAKE_MODULE_LINKER_FLAGS_RELWITHDEBINFO}")
string(REPLACE "/DNDEBUG" "" CMAKE_CXX_FLAGS_RELWITHDEBINFOANDERRORCHECKSANDASSERTS "${CMAKE_CXX_FLAGS_RELWITHDEBINFOANDERRORCHECKSANDASSERTS}")
string(REPLACE "-DNDEBUG" "" CMAKE_CXX_FLAGS_RELWITHDEBINFOANDERRORCHECKSANDASSERTS "${CMAKE_CXX_FLAGS_RELWITHDEBINFOANDERRORCHECKSANDASSERTS}")
string(APPEND CMAKE_CXX_FLAGS_RELWITHDEBINFOANDERRORCHECKSANDASSERTS " /DPERFORM_ERROR_CHECKS")

# Set C++23 standard
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_DEBUG_INFORMATION_FORMAT ProgramDatabase)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

# Print configuration details
message(STATUS "CMAKE_BUILD_TYPE: ${CMAKE_BUILD_TYPE}")
message(STATUS "CMAKE_CXX_FLAGS: ${CMAKE_CXX_FLAGS}")
message(STATUS "CMAKE_CXX_FLAGS_DEBUG: ${CMAKE_CXX_FLAGS_DEBUG}")
message(STATUS "CMAKE_CXX_FLAGS_RELEASE: ${CMAKE_CXX_FLAGS_RELEASE}")
message(STATUS "Configs: ${CMAKE_CONFIGURATION_TYPES}")

# Project type (Dynamic Library)
add_library(${PROJECT_NAME} SHARED dllmain.cpp)

# Include directories
target_include_directories(${PROJECT_NAME} PRIVATE 
    "Detours/include"
    "plog/include"
    "Tribes-Ascend-SDK"
    "Unreal-UFunction-Hook"
    "Circular-Buffer"
    "Object-Pool"
)

# Source files
target_sources(${PROJECT_NAME} PRIVATE
    dllmain.cpp
    helper.cpp
    lag_compensation.cpp
    native_hooks.cpp
    processinternal_hooks.cpp

    Unreal-UFunction-Hook/uhook.cpp
    
    # SDK Sources
    Tribes-Ascend-SDK/SDK_SOURCES/Core_functions.cpp
    Tribes-Ascend-SDK/SDK_SOURCES/Engine_functions.cpp
    # Tribes-Ascend-SDK/SDK_SOURCES/GameFramework_functions.cpp
    # Tribes-Ascend-SDK/SDK_SOURCES/GFxUI_functions.cpp
    # Tribes-Ascend-SDK/SDK_SOURCES/IpDrv_functions.cpp
    # Tribes-Ascend-SDK/SDK_SOURCES/OnlineSubsystemMcts_functions.cpp
    # Tribes-Ascend-SDK/SDK_SOURCES/PlatformCommon_functions.cpp
    # Tribes-Ascend-SDK/SDK_SOURCES/TribesGameContent_functions.cpp
    Tribes-Ascend-SDK/SDK_SOURCES/TribesGame_functions.cpp
    # Tribes-Ascend-SDK/SDK_SOURCES/UDKBase_functions.cpp
    Tribes-Ascend-SDK/SDK_SOURCES/UTGame_functions.cpp
    # Tribes-Ascend-SDK/SDK_SOURCES/WinDrv_functions.cpp
    # Tribes-Ascend-SDK/SDK_SOURCES/XAudio2_functions.cpp
)

# Linker settings
target_link_libraries(${PROJECT_NAME} PUBLIC
    "${CMAKE_SOURCE_DIR}/Detours/lib.X86/detours.lib"
)
